    /**
     * Update method called every server tick
     */
    public void update() {
        if (!initialized) {
            return;
        }

        // Get current world time from overworld
        if (server.worldMngr == null || server.worldMngr.length == 0 || server.worldMngr[0] == null) {
            return;
        }

        // Check if there are any players online
        int playerCount = server.configManager != null ? server.configManager.playerEntities.size() : 0;

        if (playerCount == 0) {
            // No players online - pause the timer
            if (!isPaused) {
                pausedTick = server.worldMngr[0].getWorldTime();
                isPaused = true;
                System.out.println("[ServerMysticManager] Paused - no players online");
            }
            return;
        } else if (isPaused) {
            // Players came back - resume
            long currentTick = server.worldMngr[0].getWorldTime();
            long pausedDuration = currentTick - pausedTick;

            // Adjust timers to account for pause
            if (gameStartTick > 0) {
                gameStartTick += pausedDuration;
                lastEventTick += pausedDuration;
            }

            isPaused = false;
            System.out.println("[ServerMysticManager] Resumed - players online: " + playerCount);
        }

        long currentTick = server.worldMngr[0].getWorldTime();

        // Initialize game start tick
        if (gameStartTick == 0) {
            gameStartTick = currentTick;
            lastEventTick = gameStartTick;
            scheduleNextEvent();
            System.out.println("[ServerMysticManager] Horror sequence started at tick " + gameStartTick);
        }

        // Handle final sequence
        if (finalSequenceActive) {
            updateFinalSequence(currentTick);
            return;
        }

        // Check if it's time to trigger next event
        long ticksSinceLastEvent = currentTick - lastEventTick;

        if (ticksSinceLastEvent >= nextEventDelayTicks) {
            triggerNextEvent();
            lastEventTick = currentTick;
            scheduleNextEvent();
            saveState(); // Save after each event
        }
    }

    private void scheduleNextEvent() {
        int minDelayTicks = 0;
        int maxDelayTicks = 0;

        // 20 ticks = 1 second, 1200 ticks = 1 minute
        switch (currentStage) {
            case 1:
                minDelayTicks = 50 * 20; // 50 seconds
                maxDelayTicks = 80 * 20; // 80 seconds
                break;
            case 2:
                minDelayTicks = 60 * 20; // 60 seconds
                maxDelayTicks = 90 * 20; // 90 seconds
                break;
            case 3:
                minDelayTicks = 30 * 20; // 30 seconds
                maxDelayTicks = 50 * 20; // 50 seconds
                break;
            case 4:
                minDelayTicks = 40 * 20; // 40 seconds
                maxDelayTicks = 70 * 20; // 70 seconds
                break;
        }

        // Apply speed multiplier
        minDelayTicks = (int)(minDelayTicks / speedMultiplier);
        maxDelayTicks = (int)(maxDelayTicks / speedMultiplier);

        // Prevent division by zero
        if (maxDelayTicks <= minDelayTicks) {
            maxDelayTicks = minDelayTicks + 1;
        }

        nextEventDelayTicks = minDelayTicks + random.nextInt(maxDelayTicks - minDelayTicks);

        System.out.println("[ServerMysticManager] Next event scheduled in " + (nextEventDelayTicks / 20) + " seconds (Stage " + currentStage + ")");
    }

    public void triggerNextEvent() {
        List<MysticEvent> currentEvents = getCurrentStageEvents();

        if (currentEvents.isEmpty()) {
            return;
        }

        // Filter events that can still be triggered
        List<MysticEvent> availableEvents = new ArrayList<MysticEvent>();
        for (MysticEvent event : currentEvents) {
            if (event.canTrigger()) {
                availableEvents.add(event);
            }
        }

        // If no available events, progress to next stage
        if (availableEvents.isEmpty() && currentStage < 4) {
            currentStage++;
            System.out.println("[ServerMysticManager] Progressing to Stage " + currentStage);
            scheduleNextEvent();
            return;
        }

        // If no available events and we're on stage 4 - start final sequence
        if (availableEvents.isEmpty() && currentStage == 4) {
            startFinalSequence();
            return;
        }

        // Choose random event from available
        int index = random.nextInt(availableEvents.size());
        MysticEvent event = availableEvents.get(index);

        // Increment trigger counter
        event.trigger();

        System.out.println("[ServerMysticManager] Triggering event: " + event.name + " (Stage " + currentStage + ")");

        // Broadcast event to all players
        broadcastEventToAllPlayers(event.name, currentStage, 1.0f);
    }

    private List<MysticEvent> getCurrentStageEvents() {
        switch (currentStage) {
            case 1: return stage1Events;
            case 2: return stage2Events;
            case 3: return stage3Events;
            case 4: return stage4Events;
            default: return stage1Events;
        }
    }

    private void startFinalSequence() {
        finalSequenceActive = true;
        finalSequenceStartTick = server.worldMngr[0].getWorldTime();
        finalEventCounter = 0;

        System.out.println("[ServerMysticManager] FINAL SEQUENCE STARTED!");

        // Broadcast initial final sequence events
        broadcastEventToAllPlayers("gdi_glitch_screen", 4, 2.0f);
        broadcastEventToAllPlayers("white_noise", 4, 1.5f);
    }

    private void updateFinalSequence(long currentTick) {
        long ticksSinceStart = currentTick - finalSequenceStartTick;

        // Every 5 seconds (100 ticks), trigger intense events
        if (ticksSinceStart % 100 == 0 && finalEventCounter < 20) {
            finalEventCounter++;

            String[] finalEvents = {
                "gdi_glitch_screen",
                "gdi_invert_screen",
                "screamer_interface",
                "chat_spam",
                "death_chat",
                "bsod",
                "white_noise"
            };

            String event = finalEvents[random.nextInt(finalEvents.length)];
            broadcastEventToAllPlayers(event, 4, 2.0f);
        }

        // After 2 minutes (2400 ticks), trigger final crash
        if (ticksSinceStart >= 2400) {
            System.out.println("[ServerMysticManager] FINAL CRASH!");
            broadcastEventToAllPlayers("final_crash", 4, 3.0f);
            finalSequenceActive = false;
        }
    }

    /**
     * Broadcast horror event to all connected players
     */

        if (server.configManager == null) {
            return;
        }

        System.out.println("[ServerMysticManager] Broadcasting event '" + eventName + "' to all players");

        // Handle server-side effects for events that need server authority
        handleServerSideEffects(eventName, stage, intensity);

        // Create and broadcast packet
        Packet250HorrorSync packet = new Packet250HorrorSync(eventName, stage, intensity);
        packet.senderName = "SERVER";
        packet.targetPlayer = ""; // Broadcast to all

        server.configManager.sendPacketToAllPlayers(packet);
    }

    /**
     * Process client mystic commands to maintain compatibility with single-player logic
     */
    public void processClientCommand(EntityPlayerMP player, String command) {
        System.out.println("[ServerMysticManager] Processing client command: " + command + " from " + player.username);
        
        // Handle mystic commands that need server authority
        if (command.startsWith("/mst")) {
            // Trigger next mystic event (compatible with single-player)
            triggerNextEvent();
            player.playerNetServerHandler.sendPacket(new Packet3Chat("\u00a7aMystic event triggered!"));
            
        } else if (command.startsWith("/next")) {
            // Force next stage progression
            if (currentStage < 4) {
                currentStage++;
                scheduleNextEvent();
                player.playerNetServerHandler.sendPacket(new Packet3Chat("\u00a7aProgressed to Stage " + currentStage));
            } else {
                player.playerNetServerHandler.sendPacket(new Packet3Chat("\u00a7cAlready at maximum stage!"));
            }
            
        } else if (command.startsWith("/x10")) {
            // Speed up event timer (10x speed)
            speedMultiplier = 10.0f;
            player.playerNetServerHandler.sendPacket(new Packet3Chat("\u00a7aEvent speed set to 10x"));
            
        } else if (command.startsWith("/horror_info")) {
            // Show current horror status
            String info = "\u00a76Horror Status:\n" +
                         "\u00a7eStage: " + currentStage + "/4\n" +
                         "\u00a7eSpeed: " + speedMultiplier + "x\n" +
                         "\u00a7ePlayers: " + server.configManager.playerEntities.size();
            player.playerNetServerHandler.sendPacket(new Packet3Chat(info));
            
        } else if (command.startsWith("/horror_reset")) {
            // Reset horror system (admin only)
            if (server.configManager.isOp(player.username)) {
                currentStage = 1;
                gameStartTick = 0;
                lastEventTick = 0;
                speedMultiplier = 2.0f;
                finalSequenceActive = false;
                player.playerNetServerHandler.sendPacket(new Packet3Chat("\u00a7aHorror system reset!"));
            } else {
                player.playerNetServerHandler.sendPacket(new Packet3Chat("\u00a7cYou don't have permission to use this command!"));
            }
        }
    }

    private void teleportToBedrockTunnel(EntityPlayerMP player) {
        double teleportX = 10000.5;
        double teleportY = 101.0;
        double teleportZ = 10001.5;
        float yaw = 0.0F;
        float pitch = 0.0F;

        System.out.println("[ServerMysticManager] Optimized teleporting " + player.username + " to bedrock tunnel");

        if (player.playerNetServerHandler != null) {
            // Step 1: Pre-load only essential chunks (3x3 area around target)
            preloadChunksAround(player.worldObj, teleportX, teleportZ, 2);
            
            // Step 2: Perform teleport with optimized chunk loading
            player.playerNetServerHandler.teleportTo(teleportX, teleportY, teleportZ, yaw, pitch);
            
            // Step 3: Force inventory sync
            syncPlayerInventory(player);
            
            // Step 4: Schedule gradual chunk loading to prevent lag
            scheduleGradualChunkLoading(player, teleportX, teleportZ);
        }
    }
    
    /**
     * Pre-load chunks around target coordinates to prevent lag
     */
    private void preloadChunksAround(World world, double x, double z, int radius) {
        int centerChunkX = (int)x >> 4;
        int centerChunkZ = (int)z >> 4;
        
        System.out.println("[ServerMysticManager] Pre-loading chunks around " + centerChunkX + "," + centerChunkZ);
        
        // Load chunks in spiral pattern from center outward
        for (int r = 0; r <= radius; r++) {
            for (int cx = centerChunkX - r; cx <= centerChunkX + r; cx++) {
                for (int cz = centerChunkZ - r; cz <= centerChunkZ + r; cz++) {
                    if (r == 0 || cx == centerChunkX - r || cx == centerChunkX + r || 
                        cz == centerChunkZ - r || cz == centerChunkZ + r) {
                        // Load only perimeter chunks of each ring to reduce redundant loads
                        world.getChunkFromChunkCoords(cx, cz);
                    }
                }
            }
        }
    }
    
    /**
     * Sync player inventory efficiently
     */
    private void syncPlayerInventory(EntityPlayerMP player) {
        if (player.inventory == null) return;
        
        // Send inventory update packets for all slots
        for (int i = 0; i < player.inventory.mainInventory.length; i++) {
            ItemStack item = player.inventory.mainInventory[i];
            player.playerNetServerHandler.sendPacket(new Packet5PlayerInventory(player.entityId, i, item));
        }
        // Sync armor slots
        for (int i = 0; i < player.inventory.armorInventory.length; i++) {
            ItemStack item = player.inventory.armorInventory[i];
            player.playerNetServerHandler.sendPacket(new Packet5PlayerInventory(player.entityId, 100 + i, item));
        }
    }
    
    /**
     * Schedule gradual chunk loading to prevent server lag
     */
    private void scheduleGradualChunkLoading(EntityPlayerMP player, double x, double z) {
        // Use a simple HashMap to track chunk loading tasks
        if (!chunkLoadingTasks.containsKey(player.username)) {
            chunkLoadingTasks.put(player.username, new ChunkLoadingTask(player, x, z));
            System.out.println("[ServerMysticManager] Scheduled gradual chunk loading for " + player.username);
        }
    }
    
    /**
     * Process gradual chunk loading tasks
     */
    public void processChunkLoadingTasks() {
        Iterator<Map.Entry<String, ChunkLoadingTask>> iterator = chunkLoadingTasks.entrySet().iterator();
        while (iterator.hasNext()) {
            Map.Entry<String, ChunkLoadingTask> entry = iterator.next();
            ChunkLoadingTask task = entry.getValue();
            
            if (task.isComplete()) {
                iterator.remove();
                System.out.println("[ServerMysticManager] Chunk loading complete for " + task.player.username);
            } else {
                task.update();
            }
        }
    }
    
    /**
     * Simple chunk loading task class
     */
    private static class ChunkLoadingTask {
        private final EntityPlayerMP player;
        private final double targetX, targetZ;
        private int currentRadius;
        private int tickCounter;
        private final int maxRadius = 8; // Load up to 8 chunks radius
        
        public ChunkLoadingTask(EntityPlayerMP player, double x, double z) {
            this.player = player;
            this.targetX = x;
            this.targetZ = z;
            this.currentRadius = 3; // Start with already loaded chunks
            this.tickCounter = 0;
        }
        
        public void update() {
            tickCounter++;
            
            // Every 10 ticks (0.5 seconds), load next ring of chunks
            if (tickCounter % 10 == 0 && currentRadius < maxRadius) {
                currentRadius++;
                loadChunkRing(currentRadius);
                System.out.println("[ServerMysticManager] Loading chunk ring " + currentRadius + " for " + player.username);
            }
        }
        
        private void loadChunkRing(int radius) {
            int centerChunkX = (int)targetX >> 4;
            int centerChunkZ = (int)targetZ >> 4;
            
            // Load chunks in spiral pattern from center outward
            for (int cx = centerChunkX - radius; cx <= centerChunkX + radius; cx++) {
                for (int cz = centerChunkZ - radius; cz <= centerChunkZ + radius; cz++) {
                    if (cx == centerChunkX - radius || cx == centerChunkX + radius || 
                        cz == centerChunkZ - radius || cz == centerChunkZ + radius) {
                        // Load only perimeter chunks of each ring to reduce redundant loads
                        player.worldObj.getChunkFromChunkCoords(cx, cz);
                    }
                }
            }
        }
        
        public boolean isComplete() {
            return currentRadius >= maxRadius;
        }
    }
    
    // Map to track chunk loading tasks per player
    private final Map<String, ChunkLoadingTask> chunkLoadingTasks = new HashMap();

            return;

            return;

        for (Object playerObj : playerList) {
            if (!(playerObj instanceof EntityPlayerMP)) {
                continue;
            }

            EntityPlayerMP player = (EntityPlayerMP) playerObj;

            // World modification events
            if (eventName.equals("bedrock_tunnel")) {
                teleportToBedrockTunnel(player);
            }
            else if (eventName.equals("sign_spawn")) {
                spawnCreepySigns(player);
            }

                spawnSilhouette(player);
            }
            else if (eventName.equals("mirror_player")) {
                spawnMirrorPlayer(player);
            }
            else if (eventName.equals("forgotten_structures")) {
                spawnForgottenStructure(player);
            }
            else if (eventName.equals("world_erosion")) {
                erodeWorld(player);
            }
            else if (eventName.equals("chunk_distortion")) {
                distortChunk(player);
            }
            else if (eventName.equals("void_hole")) {
                createVoidHole(player);
            }
            else if (eventName.equals("dry_lightning")) {
                spawnDryLightning(player);
            }
            // Player manipulation events
            else if (eventName.equals("random_item")) {
                giveRandomItem(player);
            }
            else if (eventName.equals("inventory_swap")) {
                swapInventorySlots(player);
            }
            else if (eventName.equals("infinite_inventory")) {
                fillInventory(player);
            }
            else if (eventName.equals("time_flip")) {
                flipTime(player);
            }
            else if (eventName.equals("forced_turn")) {
                forceTurn180(player);
            }
            // Chat and message events
            else if (eventName.equals("fake_join_message")) {
                sendFakeJoinMessage(player);
            }
            else if (eventName.equals("fake_gl_error")) {
                sendFakeGLError(player);
            }
            else if (eventName.equals("death_chat")) {

    /**
     * Pre-load chunks around target coordinates to prevent lag
     */
    private void preloadChunksAround(World world, double x, double z, int radius) {
        int centerChunkX = (int)x >> 4;
        int centerChunkZ = (int)z >> 4;
        
        System.out.println("[ServerMysticManager] Pre-loading chunks around " + centerChunkX + "," + centerChunkZ);
        
        // Load chunks in spiral pattern from center outward
        for (int r = 0; r <= radius; r++) {
            for (int cx = centerChunkX - r; cx <= centerChunkX + r; cx++) {
                for (int cz = centerChunkZ - r; cz <= centerChunkZ + r; cz++) {
                    if (r == 0 || cx == centerChunkX - r || cx == centerChunkX + r || 
                        cz == centerChunkZ - r || cz == centerChunkZ + r) {
                        // Load only perimeter chunks of each ring to reduce redundant loads
                        world.getChunkFromChunkCoords(cx, cz);
                    }
                }
            }
        }
    }

    /**
     * Schedule gradual chunk loading to prevent server lag
     */
    private void scheduleGradualChunkLoading(EntityPlayerMP player, double x, double z) {
        // Use a simple HashMap to track chunk loading tasks
        if (!chunkLoadingTasks.containsKey(player.username)) {
            chunkLoadingTasks.put(player.username, new ChunkLoadingTask(player, x, z));
            System.out.println("[ServerMysticManager] Scheduled gradual chunk loading for " + player.username);
        }
    }
    
    /**
     * Process gradual chunk loading tasks
     */
    public void processChunkLoadingTasks() {
        Iterator<Map.Entry<String, ChunkLoadingTask>> iterator = chunkLoadingTasks.entrySet().iterator();
        while (iterator.hasNext()) {
            Map.Entry<String, ChunkLoadingTask> entry = iterator.next();
            ChunkLoadingTask task = entry.getValue();
            
            if (task.isComplete()) {
                iterator.remove();
                System.out.println("[ServerMysticManager] Chunk loading complete for " + task.player.username);
            } else {
                task.update();
            }
        }
    }
    
    /**
     * Simple chunk loading task class
     */
    private static class ChunkLoadingTask {
        private final EntityPlayerMP player;
        private final double targetX, targetZ;
        private int currentRadius;
        private int tickCounter;
        private final int maxRadius = 8; // Load up to 8 chunks radius
        
        public ChunkLoadingTask(EntityPlayerMP player, double x, double z) {
            this.player = player;
            this.targetX = x;
            this.targetZ = z;
            this.currentRadius = 3; // Start with already loaded chunks
            this.tickCounter = 0;
        }
        
        public void update() {
            tickCounter++;
            
            // Every 10 ticks (0.5 seconds), load next ring of chunks
            if (tickCounter % 10 == 0 && currentRadius < maxRadius) {
                currentRadius++;
                loadChunkRing(currentRadius);
                System.out.println("[ServerMysticManager] Loading chunk ring " + currentRadius + " for " + player.username);
            }
        }
        
        private void loadChunkRing(int radius) {
            int centerChunkX = (int)targetX >> 4;
            int centerChunkZ = (int)targetZ >> 4;
            
            // Load chunks in a ring pattern
            for (int cx = centerChunkX - radius; cx <= centerChunkX + radius; cx++) {
                for (int cz = centerChunkZ - radius; cz <= centerChunkZ + radius; cz++) {
                    // Only load perimeter chunks of ring
                    if (Math.abs(cx - centerChunkX) == radius || Math.abs(cz - centerChunkZ) == radius) {
                        player.worldObj.getChunkFromChunkCoords(cx, cz);
                    }
                }
            }
        }
        
        public boolean isComplete() {
            return currentRadius >= maxRadius;
        }
    }
    
    // Map to track chunk loading tasks per player
    private final Map<String, ChunkLoadingTask> chunkLoadingTasks = new HashMap();

    /**
     * Generate bedrock tunnel structure at specified coordinates
     */
    private void generateBedrockTunnel(WorldServer world, int centerX, int centerZ) {
        System.out.println("[ServerMysticManager] Generating bedrock tunnel at " + centerX + ", " + centerZ);

        // Create a 3x3 tunnel from Y=1 to Y=100
        for (int y = 1; y <= 100; y++) {
            for (int x = centerX - 1; x <= centerX + 1; x++) {
                for (int z = centerZ - 1; z <= centerZ + 1; z++) {
                    // Walls are bedrock
                    if (x == centerX - 1 || x == centerX + 1 || z == centerZ - 1 || z == centerZ + 1) {
                        world.setBlockWithNotify(x, y, z, Block.bedrock.blockID);
                    } else {
                        // Center is air
                        world.setBlockWithNotify(x, y, z, 0);
                    }
                }
            }
        }

        // Place signs with creepy messages at various heights
        String[] messages = {
            "YOU SHOULDN'T", "BE HERE", "", "",
            "TURN BACK", "WHILE YOU", "STILL CAN", "",
            "404", "NOT FOUND", "", "",
            "ERROR", "REALITY.DLL", "MISSING", "",
            "WHY ARE YOU", "RUNNING?", "", "",
            "I SEE YOU", "", "", "",
            "NO ESCAPE", "", "", ""
        };

        int[] signHeights = {10, 25, 40, 55, 70, 85, 95};
        int messageIndex = 0;

        for (int height : signHeights) {
            if (messageIndex >= messages.length) break;

            // Place sign on the wall
            int signX = centerX + 1;
            int signY = height;
            int signZ = centerZ;

            world.setBlockWithNotify(signX, signY, signZ, Block.signPost.blockID);

            TileEntitySign sign = (TileEntitySign) world.getBlockTileEntity(signX, signY, signZ);
            if (sign != null && messageIndex + 3 < messages.length) {
                sign.signText[0] = messages[messageIndex];
                sign.signText[1] = messages[messageIndex + 1];
                sign.signText[2] = messages[messageIndex + 2];
                sign.signText[3] = messages[messageIndex + 3];
                messageIndex += 4;
            }
        }

        System.out.println("[ServerMysticManager] Bedrock tunnel generated successfully");
    }

     * Give random item - EXACT COPY from client MysticManager

    private void giveRandomItem(EntityPlayerMP player) {
        if (player.inventory == null) {
            return;
        }

        // Пытаемся найти валидный предмет (100 попыток)
        for (int attempt = 0; attempt < 100; attempt++) {
            int itemId = random.nextInt(256);

            // Проверка что предмет существует
            if (Item.itemsList[itemId] == null) {
                continue;
            }

            try {
                Item item = Item.itemsList[itemId];
                int amount = 1;

                // Проверка на стакающиеся предметы
                if (item.getItemStackLimit() > 1) {
                    amount = 1 + random.nextInt(Math.min(66, item.getItemStackLimit()));

                    System.out.println("[ServerMysticManager] Gave " + player.username + " item " + itemId + " x" + amount);
                    return; // Предмет добавлен, выходим
                }
            } catch (Exception e) {
                // Пропускаем этот предмет и пробуем следующий
                continue;
            }
        }

        // Если не удалось выдать предмет, выдаем алмаз
        try {
            ItemStack fallback = new ItemStack(Item.diamond.shiftedIndex, 1, 0);
            player.inventory.addItemStackToInventory(fallback);
            System.out.println("[ServerMysticManager] Gave " + player.username + " fallback diamond");
        } catch (Exception e) {
            System.out.println("[ServerMysticManager] Failed to give item to " + player.username);
        }

    /**
     * Save mystic state to file

    private void saveState() {
        try {
            File saveFile = new File(SAVE_FILE);
            DataOutputStream out = new DataOutputStream(new FileOutputStream(saveFile));

            out.writeInt(currentStage);

    /**
     * Load mystic state from file
     */
    private void loadState() {
        try {
            File saveFile = new File(SAVE_FILE);
            if (!saveFile.exists()) {
                System.out.println("[ServerMysticManager] No saved state found, starting fresh");
                return;
            }

            DataInputStream in = new DataInputStream(new FileInputStream(saveFile));

            // Read version
            int version = in.readInt();
            if (version != 1) {
                System.out.println("[ServerMysticManager] Unknown save version, starting fresh");
                in.close();
                return;
            }

            // Read basic state
            currentStage = in.readInt();
            gameStartTick = in.readLong();
            lastEventTick = in.readLong();
            nextEventDelayTicks = in.readLong();
            finalSequenceActive = in.readBoolean();
            finalSequenceStartTick = in.readLong();
            finalEventCounter = in.readInt();

            // Read event counters
            readEventList(in, stage1Events);
            readEventList(in, stage2Events);
            readEventList(in, stage3Events);
            readEventList(in, stage4Events);

            in.close();
            System.out.println("[ServerMysticManager] State loaded - Stage " + currentStage);
        } catch (Exception e) {
            System.err.println("[ServerMysticManager] Failed to load state: " + e.getMessage());
        }
    }

    private void writeEventList(DataOutputStream out, List<MysticEvent> events) throws IOException {
        out.writeInt(events.size());
        for (MysticEvent event : events) {
            out.writeUTF(event.name);
            out.writeInt(event.repeatCount);
        }
    }

    private void readEventList(DataInputStream in, List<MysticEvent> events) throws IOException {
        int count = in.readInt();
        for (int i = 0; i < count && i < events.size(); i++) {
            String name = in.readUTF();
            int repeatCount = in.readInt();

            // Find matching event and update counter
            for (MysticEvent event : events) {
                if (event.name.equals(name)) {
                    event.repeatCount = repeatCount;
                    break;
                }
            }
        }
    }

    /**
     * Create void hole - remove blocks in sphere
     */
    private void createVoidHole(EntityPlayerMP player) {
        WorldServer world = (WorldServer) player.worldObj;
        if (world == null) return;

        int centerX = (int) player.posX + random.nextInt(20) - 10;
        int centerY = (int) player.posY - 5;
        int centerZ = (int) player.posZ + random.nextInt(20) - 10;
        int radius = 5 + random.nextInt(5);

        for (int x = centerX - radius; x <= centerX + radius; x++) {
            for (int y = centerY - radius; y <= centerY + radius; y++) {
                for (int z = centerZ - radius; z <= centerZ + radius; z++) {
                    if (y > 0 && y < 128) {
                        double dist = Math.sqrt(
                            (x - centerX) * (x - centerX) +
                            (y - centerY) * (y - centerY) +
                            (z - centerZ) * (z - centerZ)
                        );
                        if (dist <= radius) {
                            int blockId = world.getBlockId(x, y, z);
                            if (blockId != Block.bedrock.blockID) {
                                world.setBlockWithNotify(x, y, z, 0);
                            }
                        }
                    }
                }
            }
        }

        System.out.println("[ServerMysticManager] Created void hole at " + centerX + ", " + centerY + ", " + centerZ + " for " + player.username);
    }

    /**
     * Spawn dry lightning
     */
    private void spawnDryLightning(EntityPlayerMP player) {
        WorldServer world = (WorldServer) player.worldObj;
        if (world == null) return;

        int distance = 10 + random.nextInt(20);
        double angle = random.nextDouble() * Math.PI * 2;
        double x = player.posX + Math.cos(angle) * distance;
        double z = player.posZ + Math.sin(angle) * distance;
        double y = world.getHeightValue((int)x, (int)z);

        EntityLightningBolt lightning = new EntityLightningBolt(world, x, y, z);
        world.entityJoinedWorld(lightning);

        System.out.println("[ServerMysticManager] Spawned dry lightning for " + player.username);
    }

    /**
     * Flip time (day <-> night)
     */
    private void flipTime(EntityPlayerMP player) {
        WorldServer world = (WorldServer) player.worldObj;
        if (world == null) return;

        long currentTime = world.getWorldTime();
        long timeOfDay = currentTime % 24000;

        if (timeOfDay < 12000) {
            // Day -> Night
            world.setWorldTime(currentTime + (13000 - timeOfDay));
        } else {
            // Night -> Day
            world.setWorldTime(currentTime + (25000 - timeOfDay));
        }

        System.out.println("[ServerMysticManager] Flipped time for " + player.username);
    }

    /**
     * Inner class for mystic events
     */
    private static class MysticEvent {
        String name;
        int stage;
        boolean repeatable;
        int repeatCount;
        int maxRepeats;

        MysticEvent(String name, int stage) {
            this.name = name;
            this.stage = stage;
            this.repeatable = false;
            this.repeatCount = 0;
            this.maxRepeats = 0;
        }

        MysticEvent(String name, int stage, boolean repeatable) {
            this.name = name;
            this.stage = stage;
            this.repeatable = repeatable;
            this.repeatCount = 0;
            this.maxRepeats = repeatable ? 7 : 1; // Same as client: 7 repeats for repeatable events
        }

        boolean canTrigger() {
            if (!repeatable) {
                return repeatCount == 0;
            }
            return repeatCount < maxRepeats;
        }

        void trigger() {
            repeatCount++;
        }

        void reset() {
            repeatCount = 0;
        }
    }
